-- زر AimBot
local aim = false

local button = Instance.new("TextButton")
button.Parent = main
button.Size = UDim2.new(0.8,0,0,40)
button.Position = UDim2.new(0.1,0,1,-50)
button.BackgroundColor3 = Color3.fromRGB(220,220,220)

button.TextColor3 = Color3.new(1,1,1)
button.Font = Enum.Font.GothamBold
button.TextScaled = true
Instance.new("UICorner",button).CornerRadius = UDim.new(0,15)

button.MouseButton1Click:Connect(function()
	aim = not aim
	button.Text = "AimBot : "..(aim and "ON" or "OFF")
end)

-- كشف الادوار
local murderer = nil
local sheriff = nil

function updateRoles()
	murderer = nil
	sheriff = nil

	for _,v in pairs(Players:GetPlayers()) do
		if v.Character then
			if v.Backpack:FindFirstChild("Knife") or (v.Character:FindFirstChild("Knife")) then
				murderer = v.Name
			end

			if v.Backpack:FindFirstChild("Gun") or (v.Character:FindFirstChild("Gun")) then
				sheriff = v.Name
			end
		end
	end

	local innocent = "Unknown"

	roles.Text =
	"🔪 Murderer : ".. tostring(murderer) ..
	"\n\n⭐ Sheriff : ".. tostring(sheriff) ..
	"\n\n🙂 Innocent : Everyone Else"
end

-- AimBot
RunService.RenderStepped:Connect(function()
	updateRoles()

	if aim and murderer then
		local target = Players:FindFirstChild(murderer)

		if target and target.Character and target.Character:FindFirstChild("Head") then
			local cam = workspace.CurrentCamera
			cam.CFrame = CFrame.new(
				cam.CFrame.Position,
				target.Character.Head.Position
			)
		end
	end
end)
